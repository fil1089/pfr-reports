Стратегия: Ежемесячный сбор групп + загрузка аудитории в Сенлер

🎯 Задача

  • Автоматизировать ежемесячный сбор целевых сообществ по смежным нишам (психология, психосоматика, ПТСР, тревожность и т.д.)
  • Создать конвейер загрузки аудитории в Сенлер для последующих рассылок и прогревов

📅 ЧАСТЬ 1: Ежемесячный сбор групп

Принцип работы

Каждый месяц собираем свежие сообщества по 5–7 смежным темам (вращаем ниши, чтобы охватить всю воронку осознанности):

Месяц 1 — Широкая воронка (низкая осознанность):

  • Психосоматика (psychosomatics)
  • Тревожность (anxiety)
  • Панические атаки (panic_attacks)
  • Бессонница (insomnia)
  • Хроническая усталость (chronic_fatigue)

Месяц 2 — Средняя воронка (ищут решение):

  • Психологи онлайн (psychologists_online)
  • Телесная терапия (somatic_therapy)
  • Работа с травмой (trauma_therapy)
  • ПТСР (ptsd)
  • ДПДГ / EMDR (emdr_therapy)

Месяц 3 — Узкая воронка (знают методы):

  • ПФР (pfr_method)
  • Билатеральная стимуляция (bilateral_stimulation)
  • Нейропсихология (neuropsychology)
  • Фасциальная терапия (fascial_therapy)
  • Психотерапия выгорания (burnout_therapy)

Автоматизация через cron / планировщик задач

Скрипт monthly_harvest.py (создать):

#!/usr/bin/env python3
"""
Ежемесячный сбор целевых групп по заданным темам.
Запускается 1-го числа каждого месяца через cron/Task Scheduler.
"""
import subprocess
from datetime import datetime
from pathlib import Path
Ротация тем по месяцам (цикл 3 месяца)

THEMES_ROTATION = {
    1: [  # Месяцы 1, 4, 7, 10 — широкая воронка
        ("психосоматика", "psychosomatics"),
        ("тревожность", "anxiety"),
        ("панические атаки", "panic_attacks"),
        ("бессонница", "insomnia"),
        ("хроническая усталость", "chronic_fatigue"),
    ],
    2: [  # Месяцы 2, 5, 8, 11 — средняя воронка
        ("психологи онлайн", "psychologists_online"),
        ("телесная терапия", "somatic_therapy"),
        ("работа с травмой", "trauma_therapy"),
        ("ПТСР", "ptsd"),
        ("ДПДГ EMDR", "emdr_therapy"),
    ],
    0: [  # Месяцы 3, 6, 9, 12 — узкая воронка
        ("ПФР метод", "pfr_method"),
        ("билатеральная стимуляция", "bilateral_stimulation"),
        ("нейропсихология", "neuropsychology"),
        ("фасциальная терапия", "fascial_therapy"),
        ("психотерапия выгорания", "burnout_therapy"),
    ],
}
def main():
    month = datetime.now().month
    cycle = month % 3
    themes = THEMES_ROTATION[cycle]
    
    print(f"📅 Месяц {month}, цикл {cycle+1}/3")
    print(f"🎯 Темы: {[t[0] for t in themes]}\n")
    
    for ru_topic, slug in themes:
        print(f"\n{'='*60}")
        print(f"Собираю: {ru_topic} ({slug})")
        print(f"{'='*60}\n")
        
        # Шаг 1: Харвестер
        subprocess.run([
            "python", "09_vk_bot/scripts/vk_harvester.py",
            "--topic", ru_topic,
            "--output", f"09_vk_bot/data/harvested_{slug}.json"
        ], check=True)
        
        # Шаг 2: JEV-фильтр
        subprocess.run([
            "python", "09_vk_bot/scripts/jev_filter.py",
            "--input", f"09_vk_bot/data/harvested_{slug}.json",
            "--topic", ru_topic,
            "--output-json", f"09_vk_bot/data/clean_{slug}_communities.json",
            "--output-txt", f"09_vk_bot/data/clean_{slug}_ids.txt"
        ], check=True)
        
        # Шаг 3: Генерация веб-отчета
        subprocess.run([
            "python", "09_vk_bot/scripts/report_hub.py",
            "--dataset", f"09_vk_bot/data/clean_{slug}_communities.json",
            "--slug", slug,
            "--title", f"🎯 База: {ru_topic}",
            "--badge", ru_topic
        ], check=True)
    
    print("\n✅ Сбор завершен! Отчеты: https://fil1089.github.io/pfr-reports/")
if __name__ == "__main__":
    main()

📥 ЧАСТЬ 2: Загрузка аудитории в Сенлер

Стратегия сбора

После того как собраны группы, выгружаем активную аудиторию (не просто подписчиков, а тех, кто лайкает/комментирует — они теплее).

Скрипт collect_for_senler.py (создать):

#!/usr/bin/env python3
"""
Сбор активной аудитории из целевых групп для загрузки в Сенлер.
Использует режим --mode active (лайки + комментарии за последние 30 дней).
"""
import subprocess
import json
from pathlib import Path
from datetime import datetime
DATA_DIR = Path("09_vk_bot/data")
AUDIENCES_DIR = DATA_DIR / "audiences"
SENLER_DIR = DATA_DIR / "senler_import"
SENLER_DIR.mkdir(exist_ok=True)
def collect_active_users(ids_file: Path, output_prefix: str):
    """Выгружает активных пользователей из групп."""
    print(f"📊 Собираю активную аудиторию из {ids_file.name}...")
    
    subprocess.run([
        "python", "09_vk_bot/scripts/audience_collector.py",
        "--targets-file", str(ids_file),
        "--mode", "active",  # Только активные (лайки/комментарии)
        "--output-dir", str(AUDIENCES_DIR)
    ], check=True)
    
    # Находим свежий файл аудитории
    latest = max(AUDIENCES_DIR.glob("active_audience_*.txt"), key=lambda p: p.stat().st_mtime)
    
    # Копируем в папку для Сенлер с понятным названием
    senler_file = SENLER_DIR / f"{output_prefix}_{datetime.now():%Y-%m}.txt"
    senler_file.write_text(latest.read_text(encoding="utf-8"), encoding="utf-8")
    
    print(f"✅ Сохранено для Сенлер: {senler_file}")
    print(f"📈 Пользователей: {len(senler_file.read_text().splitlines())}")
    
    return senler_file
def main():
    """Собирает аудиторию из всех баз текущего месяца."""
    ids_files = sorted(DATA_DIR.glob("clean_*_ids.txt"))
    
    if not ids_files:
        print("⚠️ Нет файлов clean_*_ids.txt. Сначала запусти monthly_harvest.py")
        return
    
    print(f"📂 Найдено баз: {len(ids_files)}\n")
    
    all_users = set()
    
    for ids_file in ids_files:
        slug = ids_file.stem.replace("clean_", "").replace("_ids", "")
        senler_file = collect_active_users(ids_file, slug)
        
        # Собираем уникальных пользователей для дедупликации
        users = set(senler_file.read_text().splitlines())
        all_users.update(users)
        print(f"   Уникальных в {slug}: {len(users)}\n")
    
    # Сохраняем объединенную базу (все уникальные пользователи)
    merged_file = SENLER_DIR / f"merged_active_{datetime.now():%Y-%m}.txt"
    merged_file.write_text("\n".join(sorted(all_users)), encoding="utf-8")
    
    print(f"\n{'='*60}")
    print(f"✅ ИТОГО уникальных пользователей: {len(all_users)}")
    print(f"📁 Объединенная база: {merged_file}")
    print(f"📂 Все файлы для Сенлер: {SENLER_DIR}/")
    print(f"{'='*60}")
if __name__ == "__main__":
    main()

#### Вариант 2: API Сенлер (автоматизация)

import requests
SENLER_API_KEY = "ВАШ_КЛЮЧ_ИЗ_ЛИЧНОГО_КАБИНЕТА"
VK_GROUP_ID = 236274472  # ID группы psiholog_firsov
def upload_to_senler(user_ids: list[int], segment_name: str):
    """Загружает список VK ID в базу Сенлер."""
    url = "https://senler.ru/api/subscribers/add"
    
    response = requests.post(url, json={
        "vk_group_id": VK_GROUP_ID,
        "subscriber_ids": user_ids,
        "segment": segment_name,  # Сегмент для фильтрации
        "tags": ["активная_аудитория", "конкуренты"],
    }, headers={"Authorization": f"Bearer {SENLER_API_KEY}"})
    
    return response.json()

📊 Метрики и контроль

Отслеживаем в Сенлер:

  • Open Rate (% открывших) — норма 20–30% для холодной базы, 40–60% для прогретой
  • CTR (% кликов по кнопкам) — норма 5–15%
  • Конверсия в запись — цель 2–5% от импортированной базы за 30 дней
